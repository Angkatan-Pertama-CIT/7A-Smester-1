TIPE DATA:tipe data adalah jenis data yang digunakan untuk menentukan bentuk atau jenis suatu nilai dalam program.

chart=Data yang berisi satu karakter atau satu huruf 
contoh:'A','B','C'

string=Data yang berisi kumpulan kalimat,karakter,atau kata
contoh:"huruf 123"

integer=Data berupa bilangan bulat tanpa koma
contoh:100

float=Data berupa bilangan yang memiliki angka dibelakang koma
contoh:45,30

boolean=Data yang hanya memiliki dua nilai,yaitu benar atau salah(true/false)
contoh:true/false

list=kumpulan data
contoh:[1,2,3]

OPERATOR:tanda yang digunakan untuk melakukan perhitungan atau perbandingan dalam program
contoh:>= lebih besar
       <= lebih kecil
       == sama dengan
       && dan
       || atau
       ! tidak sama dengan
